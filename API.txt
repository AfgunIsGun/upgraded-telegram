from gradio_client import Client, handle_file

client = Client("Wan-AI/Wan2.2-Animate")
result = client.predict(
	ref_img=handle_file('https://raw.githubusercontent.com/gradio-app/gradio/main/test/test_files/bus.png'),
	video={"video":handle_file('https://github.com/gradio-app/gradio/raw/main/demo/video_component/files/world.mp4')},
	model_id="wan2.2-animate-move",
	model="wan-pro",
	api_name="/predict"
)
print(result)

	});
